## Usage

```python
from .experiment_docs import check_experiment_docs, get_experiment_gate

# Check completeness
result = check_experiment_docs("LAB-063")
if not result.complete:
    print(result.missing)

# Create run document
gate = get_experiment_gate()
gate.create_run_document(
    experiment_path="/path/to/LAB-063",
    run_id=1,
    commands=["go test ./...", "go run test.go"],
    outputs=["Test output captured..."],
    result="PASS"
)
```
